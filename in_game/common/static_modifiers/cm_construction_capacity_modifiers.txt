# Grants a very large bonus to how many buildings can be under construction
# at once in a single location (vanilla base is normally 1, or 2 for higher-
# ranked locations). 1000 is effectively unlimited: no location will ever
# realistically stage that many simultaneous constructions, since gold,
# workforce, and building-slot capacity remain real throttles elsewhere.
cm_unlimited_construction_slots_modifier = {
	game_data = {
		category = location
	}
	max_constructions_at_same_time = 1000
}
