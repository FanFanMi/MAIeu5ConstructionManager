# Grants a large bonus to how many buildings can be under construction at once
# in a single location (vanilla base is normally 1, or 2 for higher-ranked
# locations). 20 is far beyond any realistic per-location need, but avoids
# flooding the construction queue's per-cycle GUI-widget verification pass,
# which has a limited number of rounds to instantiate everything staged in
# a given month (see cm_q_sync_check in cm_hidden_window_scripted_gui.txt).
cm_unlimited_construction_slots_modifier = {
	game_data = {
		category = location
	}
	max_constructions_at_same_time = 20
}
